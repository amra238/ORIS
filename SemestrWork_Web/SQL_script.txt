DROP TABLE IF EXISTS tours;

CREATE TABLE tours (
    id SERIAL PRIMARY KEY,
    -- Основная информация
    tour_name VARCHAR(255) NOT NULL,
    departure_city VARCHAR(100) NOT NULL,
    arrival_city VARCHAR(100) NOT NULL,
    departure_date DATE NOT NULL,
    end_date DATE NOT NULL,
    nights_count INTEGER NOT NULL CHECK (nights_count > 0),
    people_count INTEGER NOT NULL CHECK (people_count > 0),
    tour_price DECIMAL(10, 2) NOT NULL CHECK (tour_price >= 0),
    image_url VARCHAR(500), 
    
    avg_water_temp DECIMAL(4, 1),
    avg_air_temp DECIMAL(4, 1),
    general_description TEXT,
    location_description TEXT,
    hotel_name VARCHAR(255),
    beach_description TEXT,
    features TEXT,
    contacts TEXT,
    
    nearby_attractions TEXT,
    hotel_facilities TEXT,
    adult_pools_count INTEGER,
    children_pools_count INTEGER,
    beach_info TEXT,
    contact_info TEXT,
    
    rating INTEGER DEFAULT 5 CHECK (rating BETWEEN 1 AND 5),
    meal_plan VARCHAR(50) DEFAULT 'Все включено',

    CONSTRAINT valid_dates CHECK (end_date > departure_date)
);
